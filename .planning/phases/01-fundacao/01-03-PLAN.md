       ```typescript
       export const http = axios.create({
         headers: {
           'User-Agent': 'SoulmaskBR-Wiki-Bot/1.0 (+https://soulmaskbr.com.br)',
         },
         timeout: 15000,
       })

       let lastRequest = 0

       http.interceptors.request.use(async (config) => {
         const now = Date.now()
         const wait = Math.max(0, 1500 - (now - lastRequest))
         if (wait > 0) await new Promise(r => setTimeout(r, wait))
         lastRequest = Date.now()
         return config
       })
       ```

    4. Create `scripts/scraper/utils/mdx-writer.ts`:
       ```typescript
       import fs from 'fs/promises'
       import path from 'path'
       import type { ScrapedContent } from './types'

       export async function writeMdx(item: ScrapedContent, contentDir: string): Promise<string> {
         const dir = path.join(contentDir, item.category)
         await fs.mkdir(dir, { recursive: true })

         const frontmatter = [
           '---',
           `title: "${item.title}"`,
           `slug: "${item.slug}"`,
           `category: "${item.category}"`,
           ...(item.subcategory ? [`subcategory: "${item.subcategory}"`] : []),
           `gameVersion: "${item.gameVersion ?? 'unknown'}"`,
           `lastScrapedAt: "${new Date().toISOString().split('T')[0]}"`,
           `sourceUrl: "${item.sourceUrl}"`,
           `published: false`,
           '---',
           '',
           item.body,
         ].join('\n')

         const filePath = path.join(dir, `${item.slug}.mdx`)
         await fs.writeFile(filePath, frontmatter, 'utf-8')
         return filePath
       }
       ```

    1. Create `scripts/scraper/fandom.ts` — Fandom wiki scraper:

       Uses Fandom MediaWiki API at `https://soulmask.fandom.com/api.php`.

       Functions:
       - `getAllPages(cmtitle: string): Promise<string[]>` — Enumerates all pages in a category using `action=query&list=categorymembers`. Handles pagination with `cmcontinue` token. Uses the rate-limited `http` client.
       - `getPageContent(title: string): Promise<string>` — Fetches page wikitext using `action=parse&page={title}&prop=wikitext&format=json`. Returns parsed wikitext.
       - `parseItemPage(title: string, wikitext: string): ScrapedContent` — Extracts structured data from wikitext. Converts wikitext to Markdown-compatible format (strip wiki templates, convert links). Sets category based on page category.
       - `scrapeFandom(options: ScraperOptions): Promise<void>` — Main entry point. Enumerates categories (Items, Masks, Guides), fetches each page, transforms to ScrapedContent, calls writeMdx.

       Category mapping for Fandom:
       - `Category:Items` -> category: 'items'
       - `Category:Masks` -> category: 'masks'
       - `Category:Guides` -> category: 'guides' (may not exist — handle gracefully)

       Slug generation: lowercase title, replace spaces with hyphens, remove special characters.

       Dry run mode: When `options.dryRun` is true, enumerate pages and log titles but do NOT fetch content or write files.

       Error handling: Wrap each page fetch in try/catch. Log errors to console.error with page title. Continue to next page on failure (do not abort entire scrape).

    3. Create `scripts/scraper/index.ts` — CLI orchestrator:

       ```typescript
       #!/usr/bin/env npx tsx
       import { scrapeFandom } from './fandom'
       import { scrapeWikiGG } from './wikigg'
       import type { ScraperOptions } from './utils/types'
       import path from 'path'

       const args = process.argv.slice(2)
       const dryRun = args.includes('--dry-run')
       const fandomOnly = args.includes('--fandom-only')
       const contentDir = path.resolve(__dirname, '../../content')

       const options: ScraperOptions = {
         dryRun,
         contentDir,
         delayMs: 1500,
       }

       async function main() {
         console.log(`Scraper starting... (dry-run: ${dryRun})`)
         console.log(`Content directory: ${contentDir}`)

         console.log('\n--- Fandom Wiki ---')
         await scrapeFandom(options)

         if (!fandomOnly) {
           console.log('\n--- wiki.gg ---')
           await scrapeWikiGG(options)
         }

         console.log('\nScraping complete.')
       }

       main().catch((err) => {
         console.error('Scraper failed:', err)
         process.exit(1)
       })
       ```

       ## Overview
       Per D-09: All wiki content follows an AI-draft + manual review pipeline before publication.

       ## Steps

       ### 1. Scrape (Automated)
       - Run `npm run scrape` to fetch English content from Fandom/wiki.gg
       - Output: MDX files in `content/{category}/` with `published: false`

       ### 2. AI Draft Translation (Semi-automated)

       - AI output replaces the English body in the MDX file

         - Gaming terminology consistency (use Brazilian gaming community terms)
         - SEO-friendly title and description
         - Correct frontmatter fields
       - Makes corrections as needed

       - Reviewer sets `published: true` in frontmatter
       - Commits and pushes — Vercel rebuilds with ISR

       title: "Nome do Item"

       - `published: true` = Reviewed and approved for display

       - Dashboard showing review status per category
