    ```typescript
    export interface ItemFrontmatter {
      title: string
      slug: string
      category: 'items' | 'masks' | 'guides' | 'patches'
      subcategory?: string
      gameVersion: string
      sourceUrl: string
      published: boolean
      lastScrapedAt?: string
      // Phase 2 additions (D-08):
      tags?: string[]
      tier?: string          // items: comum | incomum | raro | epico | lendario
      difficulty?: string    // masks only: Iniciante | Intermediario | Avancado
      image?: string         // URL or relative path
      stats?: Record<string, string | number>
      crafting?: Array<{ item: string; quantity: number; image?: string }>
      location?: string[]
    }

    // Alias for masks with same shape but semantic distinction
    export type MaskFrontmatter = ItemFrontmatter

    export interface PatchFrontmatter {
      title: string
      slug: string
      category: 'patches'
      published: boolean
      version?: string
      date?: string        // ISO date string
      gameVersion?: string
      sourceUrl?: string
    }

    export interface ContentItem<T = ItemFrontmatter> {
      frontmatter: T
      content: string      // raw MDX body
      slug: string
    }

    // URL segment -> filesystem directory mapping
    export const CATEGORY_MAP: Record<string, string> = {
      itens: 'items',
      mascaras: 'masks',
      guias: 'guides',
      'patch-notes': 'patches',
    }

    // Reverse map for breadcrumbs
    export const CATEGORY_LABELS: Record<string, string> = {
      itens: 'Itens & Crafting',
      mascaras: 'Mascaras & Skills',
      guias: 'Guias & Tutoriais',
      'patch-notes': 'Patch Notes',
    }
    ```

    **Step 4 — Create `src/lib/content.ts`:**
    Follow Pattern 1 from 02-RESEARCH.md exactly. Functions:
    - `getAllContent(category: string, publishedOnly = true): Promise<ContentItem[]>` — reads all MDX files from `content/{category}/`, parses frontmatter with gray-matter, returns sorted by title A-Z. The `category` param is the filesystem directory name ('items', 'masks', 'guides', 'patches').
    - `getContentBySlug(category: string, slug: string): Promise<ContentItem | null>` — reads single MDX file `content/{category}/{slug}.mdx`, returns frontmatter + raw body.
    - `getContentCategories(category: string): Promise<string[]>` — returns unique subcategory values from all content in a category (for filter tabs).

    Use `fs/promises` and `path.join(process.cwd(), 'content')` as base. Import `matter` from `gray-matter`. All functions are async, server-side only (no 'use client').
